1. get the files (we check out the specific commit only)

   git init frappe_docker
   cd frappe_docker
   git remote add origin https://github.com/frappe/frappe_docker.git
   git fetch --depth 1 origin 99d9a1dc385d8a0660a67b0cbbfeac3d5229e58c
   git checkout FETCH_HEAD

2. generate (or use) custom env
   - use v15 is fine, we trust the devs

3. (maybe) no need to build image

4. render compose file
   - apply overrides
   - apply custom env

5. start the project (docker compose up)

6. add site(s)
   for each site:
   1. add site to dns
   2. request ssl cert
   3. frappe bench create new site


get all active sites:
docker compose -p frappe-test-project exec backend find sites -name site_config.json | cut -f 2 -d /

see what service a container is running:
echo; docker compose -p frappe-test-project exec queue-long sh -c "for cmdline in /proc/*/cmdline; do echo \$cmdline | cut -f 3 -d / | tr '\n' ' '; cat \$cmdline | tr -c '[:print:]' ' '; echo; done" | grep -v '/proc/.*/cmdline'; echo

see what all containers within the compose project is running:
PROJECT=frappe-test-project; echo; docker compose -p $PROJECT ps --format "{{.Names}}" | sed -e "s,^$PROJECT-,," -e 's/-[0-9]\{1,\}$//' | while read CONTAINER; do echo "$CONTAINER"; docker compose -p $PROJECT exec --interactive=false "$CONTAINER" sh -c "for cmdline in /proc/*/cmdline; do echo \$cmdline | cut -f 3 -d / | tr '\n' ' '; cat \$cmdline | tr '\0' ' '; echo; done" | grep -v '/proc/.*/cmdline'; echo; done


to put a site into maintenance mode for backup and/or moving to another server:
docker compose -p project-name exec backend bench --site site-name set-maintenance-mode on

back up the data:
docker compose -p project-name exec backend bench --site site-name backup --with-files
